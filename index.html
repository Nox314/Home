class AuthSystem { constructor() { this.baseUrl = 'https://nox314server.fritz.link/api'; this.tokenKey = 'x314_session_token'; } async request(endpoint, options = {}) { const token = localStorage.getItem(this.tokenKey); const headers = { 'Content-Type': 'application/json', ...(options.headers || {}) }; if (token) { headers.Authorization = `Bearer ${token}`; } const response = await fetch(`${this.baseUrl}${endpoint}`, { ...options, headers }); let payload = {}; try { payload = await response.json(); } catch (error) { payload = {}; } if (!response.ok) { return { success: false, message: payload.message || 'Serverfehler', status: response.status, payload }; } return { success: true, ...payload }; } async register(fullname, email, password) { return this.request('/register.php', { method: 'POST', body: JSON.stringify({ fullname, email, password }) }); } async login(email, password) { const result = await this.request('/login.php', { method: 'POST', body: JSON.stringify({ email, password }) }); if (result.success && result.token) { localStorage.setItem(this.tokenKey, result.token); } return result; } async logout() { const token = localStorage.getItem(this.tokenKey); if (!token) { return { success: true, message: 'Keine Session aktiv.' }; } const result = await this.request('/logout.php', { method: 'POST' }); if (result.success) { localStorage.removeItem(this.tokenKey); } return result; } async getCurrentUser() { const token = localStorage.getItem(this.tokenKey); if (!token) { return null; } const result = await this.request('/profile.php', { method: 'GET' }); if (!result.success) { localStorage.removeItem(this.tokenKey); return null; } return result.user || null; } isLoggedIn() { return !!localStorage.getItem(this.tokenKey); } } window.auth = new AuthSystem();
